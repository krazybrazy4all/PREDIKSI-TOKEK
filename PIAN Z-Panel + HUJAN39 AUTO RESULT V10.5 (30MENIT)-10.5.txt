        )
        .toLowerCase()
        .trim()

            ' '

    }

    function setValue(
        element,
        value
    ) {

        if (!element) {
            return;
        }


        const proto =

            element instanceof
                HTMLSelectElement

                ? HTMLSelectElement.prototype

                : HTMLInputElement.prototype;


        const setter =

            Object
            .getOwnPropertyDescriptor(
                proto,
                'value'
            )
            ?.set;


        if (setter) {

            setter.call(
                element,
                value
            );

        } else {

            element.value =
                value;
        }


        element.dispatchEvent(

            new Event(
                'input',
                {
                    bubbles:
                        true
                }
            )
        );


        element.dispatchEvent(

            new Event(
                'change',
                {
                    bubbles:
                        true
                }
            )
        );
    }


    // =========================================================
    // TANGGAL ZPANEL
    //
    // contoh:
    // 10 Sep 2026 | 00:18:34
    // =========================================================

    function parsePanelDateTime(
        text
    ) {

        const match =

            String(
                text || ''
            )
            .match(

                /\b(\d{1,2})\s+(Jan|Feb|Mar|Apr|May|Jun|Jul|Aug|Sep|Oct|Nov|Dec)\s+(\d{4})\s*\|\s*(\d{1,2}):(\d{2}):(\d{2})\b/i
            );


        if (!match) {

            return null;
        }


        const months = {

            jan: 0,
            feb: 1,
            mar: 2,
            apr: 3,
            may: 4,
            jun: 5,
            jul: 6,
            aug: 7,
            sep: 8,
            oct: 9,
            nov: 10,
            dec: 11
        };


        const month =

            months[
                match[2]
                .toLowerCase()
            ];


        if (
            month ===
            undefined
        ) {

            return null;
        }


        return new Date(

            Number(
                match[3]
            ),

            month,

            Number(
                match[1]
            ),

            Number(
                match[4]
            ),

            Number(
                match[5]
            ),

            Number(
                match[6]
            )
        );
    }


    function isPanelRowInCurrentRange(
        date
    ) {

        if (

            !(
                date instanceof Date
            )

            ||

            Number.isNaN(
                date.getTime()
            )
        ) {

            return false;
        }


        const now =
            new Date();


        const start =
            new Date(

                now.getFullYear(),

                now.getMonth(),

                now.getDate(),

                0,
                0,
                0,
                0
            );


        return (

            date >= start

            &&

            date <= now
        );
    }

    function getResultRows() {

        return [

    function getMarketFromRow(
        row
    ) {

        const marketCell =

            row.querySelector(
                'td.text-uppercase'
            );


        if (marketCell) {

            const text =
                compact(
                    marketCell.textContent
                );


            const exact =

                PASARAN.find(

                    market =>
                        compact(
                            market
                        ) ===
                        text
                );


            if (exact) {

                return exact;
            }
        }


        const image =

            row.querySelector(
                'img[src*="icon-market/"]'
            );


        if (image) {

            const source =

                String(

                    image.getAttribute(
                        'src'
                    )

                    ||

                    ''
                );


            const match =

                source.match(

                    /icon-market\/([^/?]+)\.(?:png|jpg|jpeg|webp)/i
                );


            if (match) {

                const candidate =

                    match[1]
                    .toLowerCase();


                if (

                    PASARAN.includes(
                        candidate
                    )

                ) {

                    return candidate;
                }
            }
        }


        const rowText =

            compact(
                row.textContent
            );


        return (

            PASARAN.find(

                market =>

                    rowText.includes(
                        compact(
                            market
                        )
                    )
            )

            ||

            null
        );
    }


    function getRowId(
        row
    ) {

        const updateForm =

            row.querySelector(

                'form[action*="update-resultlotto.php"]'
            );


        if (updateForm) {

            const match =

                String(

                    updateForm
                    .getAttribute(
                        'action'
                    )

                    ||

                    ''
                )
                .match(

                    /[?&]row=(\d+)/i
                );


            if (match) {

                return match[1];
            }
        }


        const deleteLink =

            row.querySelector(

                'a[href*="delresult.php"][href*="row="]'
            );


        if (deleteLink) {

            const match =

                String(

                    deleteLink
                    .getAttribute(
                        'href'
                    )

                    ||

                    ''
                )
                .match(

                    /[?&]row=(\d+)/i
                );


            if (match) {

                return match[1];
            }
        }


        return null;
    }


    function getRowInfo(
        row
    ) {

        const market =
            getMarketFromRow(
                row
            );


        const input =

            row.querySelector(
                'input[name="updangka"]'
            );


        const dateTime =

            parsePanelDateTime(
                row.textContent
            );


        return {

            row,

            market,

            input,

            value:

                input

                    ? String(
                        input.value || ''
                    )
                    .trim()

                    : '',


            dateTime,


            dateISO:

                dateTime

                    ? dateToISO(
                        dateTime
                    )

                    : null,


            rowId:

                getRowId(
                    row
                )
        };
    }


    function getTodayPanelRows() {

        return getResultRows()

            .map(
                getRowInfo
            )

            .filter(

                item =>

                    item.market

                    &&

                    item.dateTime

                    &&

                    item.dateISO

                    &&

                    isPanelRowInCurrentRange(
                        item.dateTime
                    )
            );
    }


    function getReadyRows() {

        return getTodayPanelRows()

            .filter(

                item => {

                    if (

                        isValidNumber(
                            item.market,
                            item.value
                        )

                    ) {

                        return false;
                    }


                    return isPlaceholder(
                        item.value
                    );
                }
            );
    }

            document.querySelector(

    // =========================================================
    // HISTORY DATETIME
    // =========================================================

    function parseHistoryDateTime(
        text
    ) {

        const match =

            String(
                text || ''
            )
            .match(

                /(\d{4})-(\d{2})-(\d{2})\s*(?:\|\s*)?(\d{1,2}):(\d{2}):(\d{2})/
            );


        if (!match) {

            return null;
        }


        return {

            date:

                `${match[1]}-${match[2]}-${match[3]}`,


            hour:

                Number(
                    match[4]
                ),


            minute:

                Number(
                    match[5]
                ),


            second:

                Number(
                    match[6]
                )
        };
    }


    function historyToDate(
        history
    ) {

        if (!history) {

            return null;
        }


        const date =

            new Date(

                `${history.date}T`

                +

                `${pad(history.hour)}:`

                +

                `${pad(history.minute)}:`

                +

                `${pad(history.second)}`
            );


        if (
            Number.isNaN(
                date.getTime()
            )
        ) {

            return null;
        }


        return date;
    }


    // =========================================================
    // EXTRACT RESULT HISTORY
    //
    // PENGAMAN:
    // 1 tanggal harus sama
    // 2 jam History tidak boleh di masa depan
    // 3 slot TOTO/5D/KINGKONG harus sama
    // 4 digit wajib benar
    // 5 pilih result terbaru yang valid
    // =========================================================

    function extractResultFromHistoryHTML(
        html,
        market,
        panelDate,
        config
    ) {

        if (
            !config
            ||
            !config.code
        ) {

            return null;
        }


        const doc =

            new DOMParser()
            .parseFromString(
                html,
                'text/html'
            );


        const now =
            new Date();


        const candidates =
            [];


        const tables =

            [
                ...doc.querySelectorAll(
                    'table'
                )
            ];


        for (
            const table
            of tables
        ) {

            const headerRow =

                table.querySelector(
                    'thead tr'
                )

                ||

                table.querySelector(
                    'tr'
                );


            if (!headerRow) {

                continue;
            }


            const headers =

                [
                    ...headerRow.querySelectorAll(
                        'th,td'
                    )
                ]
                .map(

                    cell =>

                        normalize(
                            cell.textContent
                        )
                );


            const dateIndex =

                headers.findIndex(

                    header =>

                        header ===
                            'tanggal'

                        ||

                        header ===
                            'date'
                );


            let numberIndex =

                headers.findIndex(

                    header =>

                        header ===
                            'nomor'

                        ||

                        header ===
                            'number'
                );


            if (
                numberIndex ===
                -1
            ) {

                numberIndex =

                    headers.findIndex(

                        header => {

                            const h =
                                compact(
                                    header
                                );


                            return (

                                h ===
                                    'nomor1'

                                ||

                                h ===
                                    'number1'

                                ||

                                h ===
                                    'prize1'

                                ||

                                h ===
                                    'firstprize'
                            );
                        }
                    );
            }


            if (

                dateIndex ===
                    -1

                ||

                numberIndex ===
                    -1
            ) {

                continue;
            }


            const rows =

                [
                    ...table.querySelectorAll(
                        'tbody tr'
                    )
                ];


            for (
                const row
                of rows
            ) {

                const cells =

                    [
                        ...row.querySelectorAll(
                            'td'
                        )
                    ];


                if (

                    !cells[
                        dateIndex
                    ]

                    ||

                    !cells[
                        numberIndex
                    ]
                ) {

                    continue;
                }


                const historyTime =

                    parseHistoryDateTime(

                        cells[
                            dateIndex
                        ].textContent
                    );


                if (!historyTime) {

                    continue;
                }


                // =============================================
                // TANGGAL HARUS SAMA DENGAN PANEL
                // =============================================

                if (

                    historyTime.date !==
                    panelDate

                ) {

                    continue;
                }


                const fullDate =

                    historyToDate(
                        historyTime
                    );


                if (!fullDate) {

                    continue;
                }


                // =============================================
                // JANGAN AMBIL RESULT WAKTU MASA DEPAN
                // =============================================

                if (
                    fullDate >
                    now
                ) {

                    console.warn(

                        '[PIAN105 FUTURE BLOCK]',

                        market,

                        historyTime
                    );


                    continue;
                }


                // =============================================
                // SLOT JAM KHUSUS
                // =============================================

                if (

                    Number.isInteger(
                        config.slotHour
                    )

                    &&

                    historyTime.hour !==
                        config.slotHour
                ) {

                    continue;
                }


                const numberText =

                    String(

                        cells[
                            numberIndex
                        ].textContent

                        ||

                        ''
                    )

                    .replace(
                        /\s+/g,
                        ''
                    )

                    .trim();


                const regex =

                    new RegExp(

                        `(?:^|\\D)(\\d{${config.digits}})(?:\\D|$)`
                    );


                const numberMatch =

                    numberText.match(
                        regex
                    );


                if (!numberMatch) {

                    continue;
                }


                const result =
                    numberMatch[1];


                if (

                    !isValidNumber(
                        market,
                        result
                    )

                ) {

                    continue;
                }


                candidates.push({

                    number:
                        result,

                    historyDate:
                        historyTime.date,

                    historyHour:
                        historyTime.hour,

                    historyMinute:
                        historyTime.minute,

                    historySecond:
                        historyTime.second,

                    historyTime:

                        `${pad(historyTime.hour)}:`

                        +

                        `${pad(historyTime.minute)}:`

                        +

                        `${pad(historyTime.second)}`,


                    timestamp:

                        fullDate.getTime()
                });
            }
        }


        if (
            !candidates.length
        ) {

            return null;
        }


        // result paling baru yang benar

        candidates.sort(

            (
                a,
                b
            ) =>

                b.timestamp

                -

                a.timestamp
        );


        return candidates[0];
    }

    async function runHistoryJob(
        command
    ) {

        if (

            !command

            ||

            !command.id

            ||

            !Array.isArray(
                command.targets
            )
        ) {

            return;
        }


        const numbers =
            {};


        const missing =
            {};


        const htmlCache =
            {};


        for (

            let i = 0;

            i <
            command.targets.length;

            i++
        ) {

            if (

                GM_getValue(
                    K.stop,
                    ''
                )

                ===

                command.id
            ) {

                GM_setValue(

                    K.status,

                    {
                        jobId:
                            command.id,

                        status:
                            'stopped'
                    }
                );


                return;
            }


            const target =
                command.targets[i];


            const market =
                target.market;


            const panelDate =
                target.date;


            const config =

                resolveHistoryConfig(
                    market
                );


            GM_setValue(

                K.status,

                {
                    jobId:
                        command.id,

                    status:
                        'syncing',

                    current:
                        i + 1,

                    total:
                        command.targets.length,

                    market,

                    date:
                        panelDate,

                    pool:
                        config?.name,

                    code:
                        config?.code
                }
            );


            if (

                !config

                ||

                !config.code
            ) {

                missing[
                    makeResultKey(
                        market,
                        panelDate
                    )
                ] = {

                    market,

                    date:
                        panelDate,

                    reason:
                        'mapping_tidak_tersedia'
                };


                continue;
            }


            try {

                if (

                    !htmlCache[
                        config.code
                    ]
                ) {

                    htmlCache[
                        config.code
                    ] =

                        await fetchHistoryPool(
                            config.code
                        );
                }


                const matched =

                    extractResultFromHistoryHTML(

                        htmlCache[
                            config.code
                        ],

                        market,

                        panelDate,

                        config
                    );


                if (matched) {

                    const key =

                        makeResultKey(
                            market,
                            panelDate
                        );


                    numbers[
                        key
                    ] = {

                        number:
                            matched.number,

                        date:
                            matched.historyDate,

                        hour:
                            matched.historyHour,

                        minute:
                            matched.historyMinute,

                        second:
                            matched.historySecond,

                        time:
                            matched.historyTime,

                        pool:
                            config.name,

                        code:
                            config.code
                    };


                    console.log(

                        '[PIAN105 HISTORY MATCH]',

                        market,

                        config.name,

                        config.code,

                        panelDate,

                        matched.historyTime,

                        '→',

                        matched.number
                    );


                } else {

                    missing[
                        makeResultKey(
                            market,
                            panelDate
                        )
                    ] = {

                        market,

                        date:
                            panelDate,

                        pool:
                            config.name,

                        code:
                            config.code,

                        reason:
                            'belum_ada_result_cocok'
                    };
                }


            } catch (error) {

                missing[
                    makeResultKey(
                        market,
                        panelDate
                    )
                ] = {

                    market,

                    date:
                        panelDate,

                    pool:
                        config.name,

                    code:
                        config.code,

                    reason:
                        error.message
                };


                console.error(

                    '[PIAN105 HISTORY ERROR]',

                    market,

                    config,

                    error
                );
            }


            GM_setValue(

                K.data,

                {
                    jobId:
                        command.id,

                    numbers,

                    missing,

                    updatedAt:
                        Date.now()
                }
            );


            await sleep(
                120
            );
        }


        GM_setValue(

            K.data,

            {
                jobId:
                    command.id,

                numbers,

                missing,

                updatedAt:
                    Date.now()
            }
        );


        GM_setValue(

            K.status,

            {
                jobId:
                    command.id,

                status:
                    'done',

                found:
                    Object.keys(
                        numbers
                    ).length,

                missing:
                    Object.keys(
                        missing
                    ).length
            }
        );
    }

                    history.number
                )
            ) {

                skipped++;

                continue;
            }


            // =============================================
            // DATE SAFETY
            // =============================================

            if (

                history.date !==
                    item.dateISO

            ) {

                console.error(

                    '[PIAN105 DATE BLOCK]',

                    item.market,

                    item.dateISO,

                    history.date
                );


                skipped++;

                continue;
            }


            // =============================================
            // RESULT TIME TIDAK BOLEH MASA DEPAN
            // =============================================

            const resultTime =

                new Date(

                    `${history.date}T`

                    +

                    `${pad(history.hour)}:`

                    +

                    `${pad(history.minute || 0)}:`

                    +

                    `${pad(history.second || 0)}`
                );


            if (

                Number.isNaN(
                    resultTime.getTime()
                )

                ||

                resultTime >
                    new Date()
            ) {

                console.error(

                    '[PIAN105 FUTURE BLOCK]',

                    item.market,

                    history
                );


                skipped++;

                continue;
            }


            // =============================================
            // SLOT SAFETY
            // =============================================

            const config =

                HISTORY_POOL[
                    item.market
                ];


            if (

                Number.isInteger(
                    config?.slotHour
                )

                &&

                Number(
                    history.hour
                ) !==
                    config.slotHour
            ) {

                console.error(

                    '[PIAN105 SLOT BLOCK]',

                    item.market,

                    config.slotHour,

                    history.hour
                );


                skipped++;

                continue;
            }


            // =============================================
            // JANGAN TIMPA ANGKA VALID
            // =============================================

            const currentValue =

                item.input

                    ? String(
                        item.input.value || ''
                    )
                    .trim()

                    : '';


            if (

                isValidNumber(
                    item.market,
                    currentValue
                )

            ) {

                skipped++;

                continue;
            }


            try {

                setStatus(

                    `✏️ [${i + 1}/${currentQueue.length}] `

                    +

                    `${item.market} | `

                    +

                    `${item.dateISO} ${history.time} → `

                    +

                    `${history.number}`
                );


                await updatePanelRow(
                    item,
                    history.number
                );


                success++;


                console.log(

                    '[PIAN105 UPDATE OK]',

                    item.market,

                    item.dateISO,

                    history.pool,

                    history.code,

                    history.time,

                    history.number
                );


            } catch (error) {

                failed++;


                console.error(

                    '[PIAN105 UPDATE ERROR]',

                    item.market,

                    error
                );
            }


            await sleep(
                UPDATE_DELAY
            );
        }


        running =
            false;


        setStatus(

            currentRunMode ===
                'auto'

                ?

                `🤖 AUTO SELESAI — berhasil ${success}, skip ${skipped}, gagal ${failed}`

                :

                `✅ SELESAI — berhasil ${success}, skip ${skipped}, gagal ${failed}`
        );


        refreshInfo();
    }


    // =========================================================
    // BRIDGE STATUS
    // =========================================================

    function handleHistoryStatus(
        status
    ) {

        if (

            !status

            ||

            status.jobId !==
                activeJobId
        ) {

            return;
        }


        if (

            status.status ===
                'syncing'
        ) {

            setStatus(

                `🌧️ History [${status.current}/${status.total}] `

                +

                `${status.market} → ${status.pool || '-'}`
            );


            return;
        }


        if (

            status.status ===
                'stopped'
        ) {

            running =
                false;


            setStatus(
                '⛔ Proses dihentikan'
            );


            return;
        }


        if (

            status.status ===
                'done'
        ) {

            if (

                finishedJobId ===
                    status.jobId

            ) {

                return;
            }


            finishedJobId =
                status.jobId;


            const data =

                GM_getValue(
                    K.data,
                    {}
                );


            const numbers =

                data.numbers

                ||

                {};


            setStatus(

                `✅ History cocok ${Object.keys(numbers).length}. Mulai update Z-Panel...`
            );


            processPanelUpdates(
                numbers
            );
        }
    }


    function initPanelBridge() {

        GM_addValueChangeListener(

            K.status,

            function (
                name,
                oldValue,
                newValue
            ) {

                handleHistoryStatus(
                    newValue
                );
            }
        );


        setInterval(

            function () {

                if (
                    !activeJobId
                ) {

                    return;
                }


                handleHistoryStatus(

                    GM_getValue(
                        K.status,
                        null
                    )
                );

            },

            800
        );
    }

    function refreshAutoUI() {

        const button =

            document.querySelector(
                '#pian105-auto'
            );


        const info =

            document.querySelector(
                '#pian105-auto-info'
            );


        const enabled =
            isAutoEnabled();


        const last =

            Number(

                GM_getValue(
                    K.last,
                    0
                )
            );


        const next =

            last

                ? last +
                    AUTO_INTERVAL

                : Date.now();


        if (button) {

            button.textContent =

                enabled

                    ? '🟢 AUTO 30 MENIT: ON'

                    : '⚪ AUTO 30 MENIT: OFF';


            button.style.background =

                enabled

                    ? '#2b8a3e'

                    : '#555';
        }


        if (info) {

            info.innerHTML =

                `Terakhir: <b>${last ? new Date(last).toLocaleTimeString() : '-'}</b>`

                +

                `<br>Berikutnya: <b>${enabled ? new Date(next).toLocaleTimeString() : '-'}</b>`;
        }
    }


    async function runAutomaticCheck(
        force = false
    ) {

        if (

            !isAutoEnabled()

            &&

            !force
        ) {

            return;
        }


        if (running) {

            return;
        }


        const now =
            Date.now();


        const last =

            Number(

                GM_getValue(
                    K.last,
                    0
                )
            );


        if (

            !force

            &&

            last

            &&

            now - last <
                AUTO_INTERVAL
        ) {

            return;
        }


        GM_setValue(
            K.last,
            now
        );


        refreshAutoUI();


        await startInputResults(
            'auto'
        );
    }


    function toggleAuto() {

        const enabled =
            !isAutoEnabled();

        refreshAutoUI();


        setStatus(

            enabled

                ? '🟢 AUTO 30 MENIT diaktifkan'

                : '⚪ AUTO 30 MENIT dimatikan'
        );


        if (enabled) {

            setTimeout(

                function () {

                    runAutomaticCheck(
                        false
                    );

                },

                500
            );
        }
    }
